cmake_minimum_required(VERSION 3.16)
project(rm_week3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(BUILD_APP "Build the video application (requires OpenCV and ONNX Runtime)" ON)
option(BUILD_VISION_TESTS "Build pose, covariance and configuration tests" ON)
include(CTest)
add_library(rm_state src/ekf9.cpp src/tracker.cpp src/imu_series.cpp src/world_frame.cpp src/frame_timeline.cpp)
target_include_directories(rm_state PUBLIC include)
target_compile_options(rm_state PRIVATE -Wall -Wextra -Wpedantic)
if(BUILD_TESTING)
  foreach(name ekf_synth world_frame covariance)
    add_executable(test_${name} tests/test_${name}.cpp)
    target_link_libraries(test_${name} PRIVATE rm_state)
    add_test(NAME ${name} COMMAND test_${name})
  endforeach()
endif()
if(BUILD_APP OR BUILD_VISION_TESTS)
  find_package(OpenCV REQUIRED COMPONENTS core calib3d imgproc videoio highgui dnn)
  add_library(rm_pose src/camera.cpp src/pnp_solver.cpp src/pose_estimator.cpp src/w3_config_loader.cpp src/state_visualizer.cpp)
  target_include_directories(rm_pose PUBLIC ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(rm_pose PUBLIC rm_state ${OpenCV_LIBS})
  target_compile_options(rm_pose PRIVATE -Wall -Wextra -Wpedantic)
  if(BUILD_TESTING AND BUILD_VISION_TESTS)
    add_executable(test_pose tests/test_pose.cpp)
    target_link_libraries(test_pose PRIVATE rm_pose)
    add_test(NAME pose COMMAND test_pose)
  endif()
  add_executable(evaluate_reprojection tools/evaluate_reprojection.cpp)
  target_link_libraries(evaluate_reprojection PRIVATE rm_pose)
endif()
if(BUILD_APP)
  set(ONNXRUNTIME_ROOT "$ENV{HOME}/onnxruntime" CACHE PATH "ONNX Runtime release root")
  find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h HINTS ${ONNXRUNTIME_ROOT}/include)
  find_library(ONNXRUNTIME_LIBRARY onnxruntime HINTS ${ONNXRUNTIME_ROOT}/lib)
  if(NOT ONNXRUNTIME_INCLUDE_DIR OR NOT ONNXRUNTIME_LIBRARY)
    message(FATAL_ERROR "ONNX Runtime missing. Set -DONNXRUNTIME_ROOT=/path, or explicitly disable BUILD_APP.")
  endif()
  add_executable(rm_week3 src/main_w3.cpp src/nn_detector.cpp src/traditional_detector.cpp src/number_classifier.cpp src/visualizer.cpp)
  target_include_directories(rm_week3 PRIVATE ${ONNXRUNTIME_INCLUDE_DIR})
  target_link_libraries(rm_week3 PRIVATE rm_pose ${ONNXRUNTIME_LIBRARY})
  set_target_properties(rm_week3 PROPERTIES BUILD_RPATH "${ONNXRUNTIME_ROOT}/lib")
  target_compile_options(rm_week3 PRIVATE -Wall -Wextra -Wpedantic)
endif()
